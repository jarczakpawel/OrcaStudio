cmake_minimum_required(VERSION 3.16)
project(slicer_linux_auth_browser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(slicer_linux_auth_browser LinuxAuthBrowser.cpp)
target_include_directories(slicer_linux_auth_browser PRIVATE "${REPO_ROOT}/deps_src")
target_link_libraries(slicer_linux_auth_browser PRIVATE PkgConfig::GTK3 PkgConfig::WEBKIT)
install(TARGETS slicer_linux_auth_browser RUNTIME DESTINATION ".")
install(PROGRAMS run_auth_browser.sh DESTINATION ".")
